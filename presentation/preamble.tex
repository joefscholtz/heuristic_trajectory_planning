% Comando interno para formatar o número com 3 dígitos (001, 012, etc.)
\newcommand{\formatreq}[1]{%
    \stepcounter{req#1}%
    REQ-#1-\ifnum\value{req#1}<10 00\else\ifnum\value{req#1}<100 0\fi\fi\arabic{req#1}%
}

% Comando customizado para gerar o bloco de requisito
% Uso: \makereq{INFIXO}{Título}{Subsistemas}{Tipo}{Prioridade}{Fase}{Descrição}{Critério}
\newcommand{\makereq}[8]{%
    \begin{block}{\textbf{\formatreq{#1}:} #2}
    \small
    \begin{tabular}{p{0.25\textwidth} p{0.7\textwidth}}
      \textbf{Subsistemas:} & #3 \\
      \textbf{Tipo:} & #4 \\
      \textbf{Prioridade:} & #5 \\
      \textbf{Fase:} & #6 \\
      \textbf{Descrição:} & #7 \\
      \textbf{Critério de Aceite:} & #8 \\
    \end{tabular}
    \end{block}
}
